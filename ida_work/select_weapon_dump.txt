int __thiscall TechnoClass_SelectWeapon(_TechnoClass *this, AbstractClass *target)
{
  _TechnoTypeClass *v3; // eax
  int result; // eax
  WeaponTypeClass *weap2__; // edi
  vt_AbstractTypeClass *weap1__; // ebx
  _TechnoClass *tgt_; // ebp
  int CurrentGattlingStage; // esi
  int v9; // eax
  int v10; // eax
  _BOOL1 IsWater; // bl

  v3 = this->vt->r.m.o.GetTechnoType(this);
  if ( TechnoTypeClass_IsTurretChanger(v3) && !this->vt->r.m.o.GetTechnoType(this)->t.RejoinTeamIfLimboed )
    return this->t.CurrentTurretNumber != -1 ? this->t.CurrentTurretNumber : 0;
  if ( this->vt->CanOccupyFire(this) )
    return 0;
  weap2__ = this->vt->GetWeapon(this, 1)->vt;
  if ( !weap2__ )
    return 0;
  weap1__ = this->vt->GetWeapon(this, 0)->vt;
  if ( !weap1__ || weap2__->NeverUse || !target )
    return 0;
  if ( this->t.r.m.o.InOpenTopped && this->vt->r.m.o.GetTechnoType(this)->t.SecondSpawnOffset.Z != -1 )// OpenTransportWeapon
    return this->vt->r.m.o.GetTechnoType(this)->t.SecondSpawnOffset.Z;// OpenTransportWeapon
  tgt_ = (target->field_18 & 1) != 0 ? target : 0;
  if ( !this->vt->r.m.o.GetTechnoType(this)->t.RejoinTeamIfLimboed )// IsGattling
  {
    if ( weap2__->Warhead->Airstrike )
    {
      if ( (target->iRTTIInfo[1].Create_ID)(target) == 6
        && HIBYTE(((target->iRTTIInfo[1].Create_ID)(target) == 6 ? &target[41] : 1312)->iRTTIInfo[0xE4].Create_ID) )// CanC4
      {
        v9 = (target->iRTTIInfo[1].Create_ID)(target);
        if ( *(((v9 == 6 ? target : 0)->iRTTIInfo[5].What_Am_I)(v9 == 6 ? target : 0) + 1517) )
        {                                       // ResourceDestination
          v10 = (target->iRTTIInfo[1].Create_ID)(target);
          return !*(((v10 == 6 ? target : 0)->iRTTIInfo[5].What_Am_I)(v10 == 6 ? target : 0) + 0x5EC);
        }
        return 1;
      }
    }
    else
    {
      if ( *(weap1__[1].GetArrayIndex + 347) && tgt_ && tgt_->vt->r.m.o.a.What_Am_I(tgt_) == eBuilding )
        return 1;
      if ( weap2__->DrainWeapon
        && tgt_
        && HIBYTE(tgt_->vt->r.m.o.GetTechnoType(tgt_)->t.Prerequisite.__.incrementAmount)
        && !this->t.DrainTarget
        && !HouseClass::Is_Ally(this->t.OwningPlayer, tgt_) )
      {
        return 1;
      }
      if ( weap2__->AreaFire && this->vt->r.m.o.GetCurrentMission(this) == mi_Unload )
        return 1;
      if ( this->vt->r.m.o.a.What_Am_I(this) == eBuilding && BYTE1(this[1].t.CurrentGattlingStage) )
        return 1;
      if ( (target->iRTTIInfo[1].Create_ID)(target) == eBuilding
        && HouseClass_Belongs_To_Ally(this->t.OwningPlayer, target)
        && weap2__->Warhead->ElectricAssault
        && BYTE1(target[41].iRTTIInfo[0xE4].Create_ID) )
      {
        return 1;                               // Overpowerable
      }
      if ( this->vt->r.m.o.a.What_Am_I(this) == eAircraft && BYTE2(this[1].t.AirstrikeTimer.TimeStart) )// BeingProduced
        return 1;
      if ( (target->iRTTIInfo[1].Create_ID)(target) == 11
        && (LODWORD(CONTAINING_RECORD(target, CellClass, a)->RadLevel) != 2 && (target->iRTTIInfo[3].Release)(target)
         || (CONTAINING_RECORD(target, CellClass, a)->field_144 & 0x100) != 0
         && this->vt->r.m.o.GetTechnoType(this)->t.ImmuneToPsionicWeapons)
        && !(target->iRTTIInfo[3].What_Am_I)(target)
        && this->vt->r.m.o.GetTechnoType(this)->t.gap_668 == 2 )// Naval
      {
        return 1;
      }
      if ( tgt_ && *(&weap2__->Warhead->Verses.None + tgt_->vt->r.m.o.GetTechnoType(tgt_)->t.o.Armor) != 0.0 )
      {
        if ( *(weap1__[1].GetArrayIndex + tgt_->vt->r.m.o.GetTechnoType(tgt_)->t.o.Armor + 0x14) == 0.0 )
          return 1;
        IsWater = *((tgt_->vt->r.m.o.GetCell)(tgt_) + offsetof(CellClass, LandType)) == 2
               || *((tgt_->vt->r.m.o.GetCell)(tgt_) + 236) == 6;
        if ( tgt_->vt->r.m.o.a.InAir(tgt_) )
          IsWater = 0;
        if ( !tgt_->t.r.m.o.OnBridge && IsWater )
        {
          result = (this->vt->SelectNavalTargeting)(this, target);
          if ( result != -1 )
            return result;
          return 0;
        }
        if ( !tgt_->vt->r.m.o.a.InAir(tgt_) && this->vt->r.m.o.GetTechnoType(this)->t.gap_668 == 2 )
          return 1;
        if ( weap2__->Projectile->AA && tgt_->vt->r.m.o.a.InAir(tgt_) )
          return 1;
      }
    }
    return 0;
  }
  CurrentGattlingStage = this->t.CurrentGattlingStage;
  if ( weap2__->Projectile->AA && tgt_ && tgt_->vt->r.m.o.a.InAir(tgt_) )
    return 2 * CurrentGattlingStage + 1;
  else
    return 2 * CurrentGattlingStage;
}
